# Ejemplo de .htaccess para publicar la versión estática en httpdocs/
# Copia estas líneas en el .htaccess de la raíz (después de hacer copia de
# seguridad del actual). Requiere mod_alias y mod_rewrite en Apache.

# 1) Servir index.html como portada (por delante de index.php)
DirectoryIndex index.html index.php

# 2) Redirecciones 301 desde las URLs antiguas de Joomla
Redirect 301 /index.php/bienvenida /bienvenida.html
Redirect 301 /index.php/servicios /servicios.html
Redirect 301 /index.php/consejos /consejos.html
Redirect 301 /index.php/ubicacion /contacto.html
Redirect 301 /index.php/galeria /galeria.html
Redirect 301 /index.php/2-uncategorised/5-politica-de-cookies /politica-de-cookies.html

# 3) Página 404 propia (opcional)
# ErrorDocument 404 /index.html

# 4) Compresión y caché para los assets estáticos (opcional)
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml
</IfModule>

<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpeg "access plus 1 month"
  ExpiresByType image/png "access plus 1 month"
  ExpiresByType image/svg+xml "access plus 1 month"
  ExpiresByType text/css "access plus 1 week"
  ExpiresByType application/javascript "access plus 1 week"
  ExpiresByType font/woff2 "access plus 1 year"
</IfModule>
